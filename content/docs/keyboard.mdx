## Usage

### Basic Example

```tsx
import Keyboard from "@/components/custom/Keyboard/Keyboard";

export default function Page() {
  return <Keyboard />;
}
```

### Light Theme

```tsx
<Keyboard theme="light" />
```

### Custom Styling

```tsx
<Keyboard className="max-w-3xl shadow-xl" />
```

### Custom Layout

```tsx
const customLayout = [
  [{ label: "Q" }, { label: "W" }, { label: "E" }, { label: "R" }],
  // ... more rows
];

<Keyboard layout={customLayout} />;
```

## Props

| Prop        | Type                | Default        | Description            |
| ----------- | ------------------- | -------------- | ---------------------- |
| `layout`    | `KeyboardKey[][]`   | MacBook layout | Custom keyboard layout |
| `theme`     | `'dark' \| 'light'` | `'dark'`       | Color theme            |
| `className` | `string`            | -              | Additional CSS classes |

## KeyboardKey Interface

```tsx
interface KeyboardKey {
  label?: string; // Main key label
  subLabel?: string; // Secondary label (top)
  icon?: React.ReactNode; // Custom icon
  wide?: boolean; // Double width key
  space?: boolean; // Extra wide space key
}
```

## Examples

### Minimal Layout

```tsx
const minimalLayout = [
  [{ label: "Q" }, { label: "W" }, { label: "E" }, { label: "R" }],
  [{ label: "A" }, { label: "S" }, { label: "D" }, { label: "F" }],
];

<Keyboard layout={minimalLayout} theme="light" />;
```

### With Icons

```tsx
import { Command, Option } from "lucide-react";

const iconLayout = [
  [
    { icon: <Command className="w-4 h-4" /> },
    { label: "Space", wide: true },
    { icon: <Option className="w-4 h-4" /> },
  ],
];

<Keyboard layout={iconLayout} />;
```
